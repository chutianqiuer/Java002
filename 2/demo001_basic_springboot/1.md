# 快速启动：

cd /workspace/2/springboot-projects

# 启动数据库（已运行）
docker compose up -d

# 初始化数据库（已执行）
docker exec -i springboot-mysql mysql -uroot -proot123 < sql/schema.sql

# 运行前台服务
java -jar web/target/web-1.0.0.jar

# 新终端运行后台管理
java -jar admin/target/admin-1.0.0.jar

# 登录成功
curl -X POST http://localhost:8080/api/auth/login \
-H "Content-Type: application/json" \
-d '{"username":"admin","password":"admin123"}'

# 返回 token
{"code":200,"message":"操作成功","data":{"token":"eyJ...","username":"admin",...}}

# 商品列表（公开接口，无需认证）
curl http://localhost:8080/api/product/page

默认账号：admin / admin123





● 以下是所有接口的测试命令：

  # ========== 认证接口 ==========
  # 登录
  curl -X POST http://localhost:8080/api/auth/login \
    -H "Content-Type: application/json" \
    -d '{"username":"admin","password":"admin123"}'

  # 退出登录
  curl -X POST http://localhost:8080/api/auth/logout \
    -H "Authorization: Bearer <token>"

  # 获取当前用户
  curl http://localhost:8080/api/auth/current \
    -H "Authorization: Bearer <token>"


  # ========== 用户接口 ==========
  # 用户注册
  curl -X POST http://localhost:8080/api/user/register \
    -H "Content-Type: application/json" \
    -d '{"username":"test","password":"123456","realName":"测试用户"}'

  # 获取用户详情
  curl http://localhost:8080/api/user/1 \
    -H "Authorization: Bearer <token>"

  # 用户分页列表
  curl "http://localhost:8080/api/user/page?page=1&pageSize=10" \
    -H "Authorization: Bearer <token>"

  # 更新用户
  curl -X PUT http://localhost:8080/api/user \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"id":1,"realName":"新名称"}'

  # 删除用户
  curl -X DELETE http://localhost:8080/api/user/2 \
    -H "Authorization: Bearer <token>"

  # 修改用户状态
  curl -X PATCH http://localhost:8080/api/user/1/status?status=0 \
    -H "Authorization: Bearer <token>"


  # ========== 商品接口 ==========
  # 创建商品
  curl -X POST http://localhost:8080/api/product \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"name":"iPhone 15","price":6999.00,"stock":100,"status":1}'

  # 获取商品详情
  curl http://localhost:8080/api/product/1 \
    -H "Authorization: Bearer <token>"

  # 商品分页列表（支持搜索）
  curl "http://localhost:8080/api/product/page?name=iPhone&page=1&pageSize=10" \
    -H "Authorization: Bearer <token>"

  # 获取推荐商品
  curl http://localhost:8080/api/product/recommend \
    -H "Authorization: Bearer <token>"

  # 更新商品
  curl -X PUT http://localhost:8080/api/product \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"id":1,"name":"iPhone 15 Pro","price":7999.00}'

  # 删除商品
  curl -X DELETE http://localhost:8080/api/product/1 \
    -H "Authorization: Bearer <token>"


  # ========== 订单接口 ==========
  # 创建订单
  curl -X POST http://localhost:8080/api/order \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{
      "receiverName":"张三",
      "receiverPhone":"13800138000",
      "shippingAddress":"北京市朝阳区",
      "remark":"请尽快发货",
      "items":[{"productId":1,"quantity":2}]
    }'

  # 获取订单详情
  curl http://localhost:8080/api/order/1 \
    -H "Authorization: Bearer <token>"

  # 根据订单号查询
  curl http://localhost:8080/api/order/no/ORD123456 \
    -H "Authorization: Bearer <token>"

  # 订单分页列表（支持筛选）
  curl "http://localhost:8080/api/order/page?status=0&page=1&pageSize=10" \
    -H "Authorization: Bearer <token>"

  # 取消订单
  curl -X POST http://localhost:8080/api/order/1/cancel \
    -H "Authorization: Bearer <token>"

  # 支付订单
  curl -X POST "http://localhost:8080/api/order/1/pay?paymentMethod=alipay" \
    -H "Authorization: Bearer <token>"

  # 发货
  curl -X POST http://localhost:8080/api/order/1/ship \
    -H "Authorization: Bearer <token>"

  # 确认收货
  curl -X POST http://localhost:8080/api/order/1/confirm \
    -H "Authorization: Bearer <token>"


  # ========== 文件上传接口 ==========
  # 上传文件
  curl -X POST http://localhost:8080/api/file/upload \
    -H "Authorization: Bearer <token>" \
    -F "file=@/path/to/file.jpg" \
    -F "dir=images"

  # 根据ID获取文件信息
  curl http://localhost:8080/api/file/1 \
    -H "Authorization: Bearer <token>"

  # 删除文件
  curl -X DELETE http://localhost:8080/api/file/1 \
    -H "Authorization: Bearer <token>"


  # ========== 后台管理接口 (端口8081) ==========
  # 后台用户分页
  curl "http://localhost:8081/admin/api/user/page?page=1&pageSize=10" \
    -H "Authorization: Bearer <token>"

  # 后台创建用户
  curl -X POST http://localhost:8081/admin/api/user \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"username":"newadmin","realName":"新管理员","status":1}'

  # 后台更新用户
  curl -X PUT http://localhost:8081/admin/api/user \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"id":1,"realName":"超级管理员"}'

  # 后台删除用户
  curl -X DELETE http://localhost:8081/admin/api/user/2 \
    -H "Authorization: Bearer <token>"

  # 后台商品分页
  curl "http://localhost:8081/admin/api/product/page?page=1&pageSize=10" \
    -H "Authorization: Bearer <token>"

  # 后台创建商品
  curl -X POST http://localhost:8081/admin/api/product \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"name":"MacBook Pro","price":12999.00,"stock":50,"status":1}'

  # 后台更新商品
  curl -X PUT http://localhost:8081/admin/api/product \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer <token>" \
    -d '{"id":1,"name":"MacBook Pro M3","price":14999.00}'

  # 后台删除商品
  curl -X DELETE http://localhost:8081/admin/api/product/2 \
    -H "Authorization: Bearer <token>"

  # 后台订单分页
  curl "http://localhost:8081/admin/api/order/page?page=1&pageSize=10" \
    -H "Authorization: Bearer <token>"

  # 后台修改订单状态
  curl -X PATCH "http://localhost:8081/admin/api/order/1/status?status=3" \
    -H "Authorization: Bearer <token>"

  说明：
  - <token> 替换为登录返回的 JWT token
  - <file.jpg> 替换为实际文件路径
  - 订单状态：0=待支付 1=已支付 2=已发货 3=已完成 4=已取消
